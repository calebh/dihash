\section{Table Filling Algorithm}

In this section we describe a table filling algorithm used to minimize
digraphs. The flavor and implementation of the algorithm are similar to
that used to minimize DFAs.

\begin{algorithm}
    \caption{Minimization Algorithm}
    \label{alg:minimization}
    \begin{algorithmic}[1] % The number tells where the line numbering should start
        \Procedure{is\_dist}{table, $a$, $bs$}
            \State \Comment{Returns \textbf{True} if $a$ is distinguishable from all nodes $b$ in $bs$}
            \For{$b \in bs$}
                \If{\textbf{not} table.is\_marked($a$, $b$)}
                    \State \textbf{return} \textbf{False}
                \EndIf
            \EndFor
            \State \Return \textbf{True}
        \EndProcedure
        \Procedure{minimize\_table}{$G$}
            \Comment{A DistinguishingTable is a table that stores parking for unordered node pairs $\{n, m\}$.}
            \State table $\gets$ new DistinguishingTable()
            \For{$(n, m) \in $ G.nodes $\times$ G.nodes}
                \If{G.label(n) $\neq$ G.label(m)}
                    \State table.mark(n, m)
                \EndIf
            \EndFor
            \State updated $\gets$ \textbf{True}
            \While{updated}
                \State updated $\gets$ \textbf{False}
                \For{$(n, m) \in $ G.nodes $\times$ G.nodes}
                    \If{\textbf{not} table.is\_marked(n, m)}
                        \For{$n' \in $ G.neighbors($n$)}
                            \If{\textsc{is\_dist}(table, $n'$, G.neighbors($m$))}
                                \State table.mark(n, m)
                                \State updated $\gets$ \textbf{True}
                            \EndIf
                        \EndFor
                    \EndIf
                \EndFor
            \EndWhile
            \State \Return table
        \EndProcedure
        \Procedure{minimize}{$G$}
            \State table $\gets$ \textsc{minimize\_table($G$)}
            \State eq\_classes $\gets$ \textsc{equivalence\_classes}($G$, table)
            \State \Return \textsc{merge}($G$, eq\_classes)
        \EndProcedure
    \end{algorithmic}
\end{algorithm}

\section{Proof of Correctness of Table Filling Algorithm}

Let $\delta(x)$ give the set of neighbors of $x$ in $G$. $\oplus$ is the logical xor operation. When we say
``$w$ distinguishes $p$ from $q$'' we mean $w \in L_G(p) \oplus w \in L_G(q)$.

\subsection{Proof of Construction of Minimal Digraph}

\begin{lemma}
\label{thm:tbl_correctness1}
$\forall w \forall p \forall q . w \in L_G(p) \oplus w \in L_G(q) \rightarrow$ The table filling algorithm finds $p$ and $q$
to be distinguished
\end{lemma}
\begin{proof}
Proof by strong induction on the length of $w$.

Base case $|w|=1$. Then the table filling algorithm
would have distinguished $p$ from $q$ during the initialization phase.

We now show the inductive step. As hypotheses, we have the following:

\begin{itemize}
    \item $w=a_1 a_2 ... a_n$
    \item $w \in L_G(p) \oplus w \in L_G(q)$
    \item $\forall w' \forall p' \forall q' . |w'| < n \rightarrow w' \in L_G(p') \oplus w' \in L_G(q') \rightarrow$ The table filling algorithm finds $p'$ and $q'$
    to be distinguished.
\end{itemize}

Our goal is to show that the table filling algorithm finds $p$ and $q$ to be distinguished.

Consider the following cases:

\begin{enumerate}
    \item The string $w'=a_1$ distinguishes $p$ from $q$. Then we can immediately apply the
    inductive hypothesis to conclude that the table filling algorithm finds $p$ and $q$
    to be distinguished.
    \item The string $w$ distinguishes $p$ from $q$, but the string $w'=a_1$ does not. Then exactly
    one of the following must be true:
    \begin{enumerate}
        \item At least one neighbor $p'$ of $p$ must be distinguished from all neighbors $q'$ of $q$ by $w'=a_2 ... a_n$.
        Consider the following cases:
        \begin{enumerate}
            \item $q$ has no neighbors. Then we will mark and therefore
            distinguish $p$ from $q$ during the first iteration of the algorithm.
            \item $q$ has at least one neighbor. Let $q'$ be one of these neighbors.
            By the inductive hypothesis, the table algorithm finds $p'$ to be
            distinguished from $q'$.
    
            This means that during the iteration of the algorithm, $p'$ and $q'$ will be marked. This allows
            the table filling algorithm to distinguish $p$ from $q$ in the next iteration of the loop.
        \end{enumerate}

        \item At least one neighbor $q'$ of $q$ must be distinguished from all neighbors $p'$ of $p$ by $w'=a_2 ... a_n$.
        We can repeat the same argument given in the previous step without loss of generality.
    \end{enumerate}
\end{enumerate}

\end{proof}

\begin{lemma}
\label{thm:tbl_correctness2}
$\forall n . \forall p \forall q.$ The table filling algorithm finds $p$ and $q$ to be distinguished on step $n$ $\rightarrow \exists w . w \in L_G(p) \oplus w \in L_G(q)$
\end{lemma}
\begin{proof}
Proof by strong induction on $n$.

Base case $n=1$. Then during the initialization of the table, we discovered that
the label of $p$ is not equal to the label of $q$. This directly shows that the string consisting
of the label of $p$ is in $L_G(p)$ but not in $L_G(q)$.

We now show the inductive step. As hypotheses, we have the following:

\begin{itemize}
    \item The table filling algorithm $p$ and $q$ to be distinguished on step $n$.
    \item $\forall m \forall p' \forall q' . m < n \rightarrow$ The table filling algorithm finds $p'$ and $q'$ to be distinguished on step $m$ $\rightarrow \exists w . w \in L_G(p') \oplus w \in L_G(q')$
\end{itemize}

Our goal is to show that $\exists w . w \in L_G(p) \oplus w \in L_G(q)$

If the table filling algorithm found $p$ and $q$ to be distinguished on step $n$,
then one of the following must be true:

\begin{enumerate}
    \item At least one neighbor $p'$ of $p$ with label $a_2$ must have been marked for
    all neighbors $q'$ of $q$. Let $a_1$ be the label of $p$. Consider the following two cases:
    \begin{enumerate}
        \item $q$ has no neighbors. Then the string $a_1 a_2$ distinguishes $p$ from $q$,
        which proves the existential in our desired goal.
        \item $q$ has at least one neighbor. Let $q'$ be one of these neighbors.
        If the pair $p'$, $q'$ are marked, this means that the algorithm found $p'$
        to be distinguished from $q'$ on some earlier step. This allows us to apply
        the inductive hypothesis to conclude $\exists w' . w' \in L_G(p') \oplus w' \in L_G(q')$.
        Since $p'$ is a neighbor of $p$ and $q'$ is a neighbor of $q$, it directly follows
        that $a_1 w' \in L_G(p) \oplus a_1 w' \in L_G(q)$. This matches our
        proof goal.
    \end{enumerate}
    \item At least one neighbor $q'$ of $q$ with label $a_2$ must have been marked for
    all neighbors $p'$ of $p'$. Without loss of generality we can repeat an identical argument
    from the previous step.
\end{enumerate}

\end{proof}

\begin{lemma}
\label{thm:tbl_correctness3}
The table filling algorithm finds $p$ and $q$ to be distinguished $\rightarrow$ $\exists n.$ The table filling
algorithm finds $p$ and $q$ to be distinguished on step $n$.
\end{lemma}
\begin{proof}
If the table filling algorithm finds $p$ and $q$ to be distinguished, then there must have been a step $n$
where $p$ and $q$ were marked.
\end{proof}

\begin{theorem}
For all vertices $p$, $q$, the table filling algorithm finds $p$ and $q$ to be
distinguished if and only if there is a $w$ that is in exactly one of $L_G(p)$ and $L_G(q)$
\end{theorem}
\begin{proof}
This immediately follows by applying Lemmas \ref{thm:tbl_correctness1}, \ref{thm:tbl_correctness2} and
\ref{thm:tbl_correctness3} appropriately. A Coq/Rocq version of mechanically verifying these lemma applications
(and properly instantiating the existentials) is available on GitHub.
\end{proof}

\subsection{Proof of Uniqueness}

\subsubsection{Existence of Bijection}

We prove this by constructing the bijection. Suppose we have a graph
$G$ minimized to $G'$ and some arbitrary minimal graph $G_{min}$ where
$L(G')=L(G_{min})$. We then show an existence of a bijection
$f : V(G') \rightarrow V(G_{min})$ when $L_{G'}(p) = L_{G_{min}}(f(p))$.

First we note that every node in $G'$ or $G_{min}$ must have a different
language. If this was not the case, then two nodes are indiscernable and
could be merged together. This would contradict the assumption that
both these graphs are minimal.

\begin{lemma}
\label{thm:bijection1}
$f$ is defined everywhere
\end{lemma}
\begin{proof}
Consider some vertex $p$ in $G$. Let $S=L_{G'}(p)$. Then $S \in L(G')$. However
$L(G')=L(G_{min})$, so $S \in L(G_{min})$. This means there must be a vertex $p'$
such that $S=L_{G_{min}}(p')$. We can then map $p \mapsto p'$ in $f$.
\end{proof}

\begin{lemma}
\label{thm:bijection2}
$f$ is well defined
\end{lemma}
\begin{proof}
$f$ is well defined if $p=q$ implies $f(p)$ = $f(q)$.

Assume $f$ is not well defined. Then there is some $p$ such that
$f(p)=p'$ and $f(p)=p''$ where $p' \neq p''$.

This means that $L_{G_{min}}(p') = L_{G_{min}}(p'')$ which means that $p'$
and $p''$ are indistinguishable. This contradicts the assumption that
$G_{min}$ is minimal.
\end{proof}

\begin{lemma}
\label{thm:bijection3}
$f$ is injective
\end{lemma}
\begin{proof}
Assume that $f$ is not injective. Then there are some states $p$ and $q$
such that $f(p)=f(q)$.

Since $L(G)=L(G_{min})$, this implies that $p$ and $q$ are indiscernable.
However our algorithm verified that $p$ and $q$ are discernable, which
is a contradiction.
\end{proof}

\begin{lemma}
\label{thm:bijection4}
$f$ is surjective
\end{lemma}
\begin{proof}
Assume $f$ is not surjective. Then there is some state $p'$ such that there
is no node $p$ such that $f(p)=p'$. This implies that $L(G') \neq L(G_{min})$,
which is a contradiction.
\end{proof}

\begin{lemma}
\label{thm:bijection_final}
Although a bijection between vertices $f$ must exist, it is not the case that
$G' \cong G_{min}$
\end{lemma}
\begin{proof}
By Lemmas \ref{thm:bijection1}, \ref{thm:bijection2}, \ref{thm:bijection3} and \ref{thm:bijection4},
$f$ must exist. However the edge structure of $G'$ and $G_{min}$ need not be the same, as the 
counterexample in Figure \ref{fig:isomorphism_counterexample} demonstrates.
\end{proof}

\subsubsection{Uniqueness by Edge Saturated Minimum}

Given a minimal graph $G$, we will now add edges to the graph until we reach a
condition of \textit{edge saturation}. We first observe that for some pair of nodes
$p$ and $q$ where $L_G(p) \supseteq L_G(q)$, for every incoming edge $(a, p)$, we can
add a new edge $(a, q)$. Due to the superset relation, this has no effect on the language
of the graph. We can repeat this process repeatedly until no more edges can be added.
We call such a graph an \textit{edge saturated} graph.

\begin{theorem}
For some graph $G$ minimized and edge saturated to $G'$ and some arbitrary minimal
edge saturated graph $G_{min}$ where $L(G')=L(G_{min})$, $G'$ is isomorphic to $G_{min}$.
\end{theorem}
\begin{proof}
By Lemma \ref{thm:bijection_final}, a bijection between the nodes of $G'$ and
$G_{min}$ must exist. We now make use of the edge saturation property to complete
a proof of isomorphism. We do this by showing that $(p,q) \in E(G')$ if and only if
$(f(p), f(q)) \in E(G_{min})$.

$\Rightarrow$ We now show that if $(p,q) \in E(G')$ then $(f(p), f(q)) \in E(G_{min})$.
Suppose that node $p$ has label $a$. Now consider the set of strings $\{aw \mid w \in L_G(q)\}$.
By the bijection of $f$, this same set of strings must also appear in the language of
$f(p)$. There are two possible ways for this entire subset to exist in the language of $f(p)$:
\begin{enumerate}
    \item $(f(p), f(q)) \in E(G_{min})$, in which case we have proved the goal.
    \item $(f(p), q') \in E(G_{min})$ where $L_{G_{min}}(q') \supseteq L_{G_{min}}(f(q))$.
    However $G_{min}$ is edge saturated, which means that the edge $(f(p), f(q))$ is also in
    $E(G_{min})$.
\end{enumerate}

$\Leftarrow$ We now show that if $(f(p), f(q)) \in E(G_{min})$, then $(p,q) \in E(G')$. This
argument is almost identical to the previous case. Suppose that node $f(p)$ has label $a$.
Now consider the set of strings $\{aw \mid w \in L_G(f(q))\}$. By the bijection of $f$, this
same set of strings must also appear in the language of $f^{-1}(f(p))=p$.
There are two possible ways for this entire susbset to exist in the language of $p$.
\begin{enumerate}
    \item $(p, q) \in E(G')$, in which case we have proved the goal.
    \item $(p, q') \in E(G')$ where $L_{G'}(q') \supseteq L_{G'}(q)$. However $G'$ is edge
    saturated, which means that $(p, q)$ is also in $E(G')$.
\end{enumerate}
\end{proof}

\subsection{Strength of Directed Vertex Language}

\begin{theorem}
Regular languages are stronger than directed vertex languages.
\end{theorem}
\begin{proof}
The language $\left\{ab\right\}$ cannot be represented by a directed vertex language.

Assume that there is some graph $G$ with language $L(G)=\left\{ab\right\}$. Then inside of
$G$ there must be a walk that starts at a vertex labeled $a$, transitions to a vertex labeled
$b$, then stops. However the mere act of visiting the vertex labeled $a$ will add $a$ to 
the language of the graph. This contradicts the assumption that $L(G)=\left\{ab\right\}$.
\end{proof}

\begin{theorem}
Every vertex language is also a regular language.
\end{theorem}
\begin{proof}
This can be accomplished by converting the graph $G$ to a NFA with $\epsilon$-transitions.
To accomplish this reduction, label all edges in $G$ with $\epsilon$. Split each vertex
into two vertices, one vertex receives all incoming edges of the original node and one
vertex emits all outgoing edges of the original node. Create an edge between this pair
of nodes and label it with the label of the original node.

At each outgoing edge node add a new epsilon transition to an accepting state.
This allows the walk through the graph to end after visiting any node. Add a starting
state to the NFA that has an epsilon transition to all other nodes in the graph
except for the accepting state.
\end{proof}
